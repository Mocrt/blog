<script setup>
import { useData, useRouter, withBase } from 'vitepress'
import { computed, onMounted, ref, onBeforeUnmount } from 'vue'

const { site } = useData()
const router = useRouter()
const homeLink = withBase('/')

// 5 秒后自动跳回首页,可点击"立即返回"提前
const seconds = ref(5)
let timer = null

onMounted(() => {
  timer = setInterval(() => {
    seconds.value -= 1
    if (seconds.value <= 0) {
      clearInterval(timer)
      router.go(homeLink)
    }
  }, 1000)
})

onBeforeUnmount(() => timer && clearInterval(timer))

function goHome() {
  clearInterval(timer)
  router.go(homeLink)
}

function goBack() {
  clearInterval(timer)
  if (window.history.length > 1) {
    window.history.back()
  } else {
    router.go(homeLink)
  }
}

const currentPath = computed(() => {
  if (typeof window === 'undefined') return ''
  return window.location.pathname + window.location.search
})
</script>

<div class="not-found">

  <p class="desc">
    你访问的地址 <code>{{ currentPath }}</code> 不存在,可能已被删除或从未存在过。
  </p>

  <div class="actions">
    <button class="btn primary" @click="goHome">返回首页</button>
    <button class="btn" @click="goBack">回到上一页</button>
  </div>

  <p class="hint">
    <span>{{ seconds }}</span> 秒后自动跳转到首页……
  </p>
</div>
